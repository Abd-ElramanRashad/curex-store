function renderDashboard(data) { document.getElementById('stat-total-sold').innerText = data.totalSoldUnits || 0; const inventoryList = data.inventory || []; renderSalesChart(inventoryList); // تصفية النواقص والمنتجات التي وصلت حد الطلب أو أقل منه const reorderAlerts = inventoryList.filter(item => { const curr = Number(item.currentBalance || 0); const hasReorderPoint = item.reorderPoint !== undefined && item.reorderPoint !== null && item.reorderPoint !== '' && String(item.reorderPoint).toLowerCase() !== 'null'; const reorder = hasReorderPoint ? Number(item.reorderPoint) : 0; // يعطي تنبيه إذا كان الرصيد الحالي أقل من أو يساوي حد الطلب، أو إذا كان الرصيد أوشك على النفاد (<= 2) بدون تحديد حد طلب return curr <= reorder || (!hasReorderPoint && curr <= 2); }); document.getElementById('stat-reorder-count').innerText = reorderAlerts.length; // 1. جدول تنبيهات حد الطلب النواقص const reorderBody = document.getElementById('reorder-table-body'); if (reorderAlerts.length === 0) { reorderBody.innerHTML = `جميع المنتجات فوق حد الطلب 👍`; } else { reorderBody.innerHTML = reorderAlerts.map(item => { const rPointDisplay = (item.reorderPoint !== null && item.reorderPoint !== undefined && item.reorderPoint !== '' && String(item.reorderPoint).toLowerCase() !== 'null') ? `${item.reorderPoint} ${item.unit || ''}` : 'غير محدد'; return ` ${item.code} ${item.name} ${item.currentBalance} ${item.unit || 'قطعة'} ${rPointDisplay} `; }).join(''); } // 2. جدول تقرير Inventory Balance الكامل const invBody = document.getElementById('inventory-table-body'); invBody.innerHTML = inventoryList.map(item => { const current = Number(item.currentBalance || 0); const hasReorderPoint = item.reorderPoint !== undefined && item.reorderPoint !== null && item.reorderPoint !== '' && String(item.reorderPoint).toLowerCase() !== 'null'; const reorderVal = hasReorderPoint ? Number(item.reorderPoint) : 0; const isNeedsReorder = current <= reorderVal || (!hasReorderPoint && current <= 2); const statusBadge = isNeedsReorder ? ` Reorder (${reorderVal})` : ` Safe (${reorderVal})`; return ` ${item.code} ${item.name} ${item.unit || ''} ${item.totalSold || 0} ${current} ${statusBadge} `}).join(''); }
