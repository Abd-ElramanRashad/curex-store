let allProducts = []; let selectedCategory = 'ALL'; // Read Excel file CureX_Database.xlsx directly from local repository async function loadProductsFromExcel() { try { const response = await fetch('./CureX_Database.xlsx'); const arrayBuffer = await response.arrayBuffer(); const data = new Uint8Array(arrayBuffer); const workbook = XLSX.read(data, { type: 'array' }); // Read first worksheet (e.g. Products or Sheet1) const firstSheetName = workbook.SheetNames[0]; const worksheet = workbook.Sheets[firstSheetName]; const rawJson = XLSX.utils.sheet_to_json(worksheet); // Map Excel Columns dynamically to Product Properties allProducts = rawJson.map((row, index) => ({ item_code: row['كود المنتج'] || row['Item Code'] || row['Code'] || `P88${index + 1}`, item_name: row['اسم المنتج'] || row['Item Name'] || row['Name'] || 'منتج طبي', category: String(row['القسم'] || row['الفئة'] || row['Category'] || 'عام').trim(), selling_price: parseFloat(row['سعر البيع'] || row['Price'] || row['سعر'] || 0), unit: row['الوحدة'] || row['Unit'] || 'قطعة', current_stock: parseInt(row['الكمية'] || row['المخزون'] || row['Stock'] || 10) })); renderProducts(); } catch (err) { console.error("Error loading Excel file:", err); document.getElementById('products-grid').innerHTML = `
⚠️ تعذر قراءة ملف CureX_Database.xlsx. يرجى التأكد من مسار الملف في المستودع.
`; } } function renderProducts() { const grid = document.getElementById('products-grid'); const search = document.getElementById('search-input').value.toLowerCase().trim(); const filtered = allProducts.filter(p => { const catText = (p.category || '').toLowerCase(); // Flexible matching logic for categories let matchesCat = false; if (selectedCategory === 'ALL') { matchesCat = true; } else if (selectedCategory === 'CHECK') { // Matches "أجهزة" or "فحص" matchesCat = catText.includes('أجهزة') || catText.includes('فحص') || catText.includes('جهاز'); } else if (selectedCategory === 'CONSUMABLE') { // Matches "استهلاكية" or "مستلزمات" or "مصرفات" matchesCat = catText.includes('استهلاك') || catText.includes('مصرف') || catText.includes('استهلاكية'); } const matchesSearch = (p.item_name && p.item_name.toLowerCase().includes(search)) || (p.item_code && p.item_code.toString().toLowerCase().includes(search)); return matchesCat && matchesSearch; }); if (filtered.length === 0) { grid.innerHTML = `

لا توجد منتجات مطابقة لهذا القسم في الشيت.

`; return; } grid.innerHTML = filtered.map(p => `
${p.item_code}

مستلزمات طبية - ${p.category || 'عام'}

${p.item_name}

${p.selling_price.toFixed(2)} ج.م / لكل ${p.unit || 'قطعة'}
`).join(''); } function setCategory(catKey, btnElement) { selectedCategory = catKey; document.querySelectorAll('.cat-btn').forEach(btn => { btn.classList.remove('bg-curexTeal', 'text-white'); btn.classList.add('bg-white', 'text-gray-600'); }); if (btnElement) { btnElement.classList.remove('bg-white', 'text-gray-600'); btnElement.classList.add('bg-curexTeal', 'text-white'); } renderProducts(); } function filterProducts() { renderProducts(); } function scrollToProducts() { document.getElementById('products-section').scrollIntoView({ behavior: 'smooth' }); } function openOrderModal(code, name, price, stock) { document.getElementById('selected-item-code').value = code; document.getElementById('selected-item-price').value = price; document.getElementById('selected-item-name').value = name; document.getElementById('order-qty').max = stock; document.getElementById('order-qty').value = 1; document.getElementById('modal-product-info').innerText = `${name} (${code}) — السعر: ${price} ج.م`; calculateTotal(); document.getElementById('order-modal').classList.remove('hidden'); } function closeModal() { document.getElementById('order-modal').classList.add('hidden'); } function calculateTotal() { const price = parseFloat(document.getElementById('selected-item-price').value) || 0; const qty = parseInt(document.getElementById('order-qty').value) || 1; document.getElementById('total-price-display').innerText = (price * qty).toFixed(2) + ' ج.م'; } async function submitOrder(e) { e.preventDefault(); alert('✅ تم تسجيل الطلب بنجاح! شكراً لثقتكم بـ CureX Healthcare Solutions.'); closeModal(); } window.onload = loadProductsFromExcel;
